# board_aim: can a player click the basket? Place a balloon, then stand at several distances and aim at several heights;
# `target` is what the crosshair sits on. Then try `use` from the best spot.
world new flat mode=survival
cmd give @s hotairballoons:hot_air_balloon
wait 10
slot 1
look 0 60
wait 8
state aim_before_place
use
wait 8
state just_placed
cmd data get entity @e[type=hotairballoons:hot_air_balloon,limit=1] Pos
cmd data get entity @e[type=hotairballoons:hot_air_balloon,limit=1] Rotation
shot placed_fp
wait 60
# the natural second click: still looking where the item was placed
state placed_look_down

cmd execute at @e[type=hotairballoons:hot_air_balloon,limit=1] run tp @s ~ ~ ~-1.2 0 0
